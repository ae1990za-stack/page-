/* Õ„«Ì… «·‰’Ê’ «· Ì ÌœŒ·Â« «·„” Œœ„ */

function escapeHTML(value) {
    return String(value ?? "").replace(/[&<>"']/g, (char) => {
        const entities = {
            "&": "&amp;",
            "<": "&lt;",
            ">": "&gt;",
            '"': "&quot;",
            "'": "&#39;"
        };

        return entities[char];
    });
}

/* «· Õﬁﬁ „‰ ·Ê‰ HEX */

function safeColor(value, fallback) {
    return /^#[0-9a-fA-F]{6}$/.test(value)
        ? value
        : fallback;
}

function validateData(data) {
    if (!data.businessName) {
        return "„‰ ›÷·ﬂ √œŒ· «”„ «·‰‘«ÿ «· Ã«—Ì.";
    }

    if (!data.headline) {
        return "„‰ ›÷·ﬂ √œŒ· «·⁄‰Ê«‰ «·—∆Ì”Ì.";
    }

    if (data.price !== "" && (
        !Number.isFinite(Number(data.price)) ||
        Number(data.price) < 0
    )) {
        return "«·”⁄— €Ì— ’ÕÌÕ.";
    }

    if (data.oldPrice !== "" && (
        !Number.isFinite(Number(data.oldPrice)) ||
        Number(data.oldPrice) < 0
    )) {
        return "«·”⁄— «·”«»ﬁ €Ì— ’ÕÌÕ.";
    }

    if (
        data.price !== "" &&
        data.oldPrice !== "" &&
        Number(data.oldPrice) < Number(data.price)
    ) {
        return "«·”⁄— «·”«»ﬁ ÌÃ» √‰ ÌﬂÊ‰ √ﬂ»— „‰ «·”⁄— «·Õ«·Ì.";
    }

    if (data.whatsapp) {
        const digits = data.whatsapp.replace(/[\s()+.-]/g, "");

        if (!/^\d{8,15}$/.test(digits)) {
            return " Õﬁﬁ „‰ —ﬁ„ Ê« ”«» Ê„› «Õ «·œÊ·….";
        }
    }

    return "";
}

/*  ‰”Ìﬁ «·√”⁄«— */

function formatPrice(value) {
    const number = Number(value);

    if (!Number.isFinite(number)) {
        return "";
    }

    return new Intl.NumberFormat("ar-EG", {
        maximumFractionDigits: 2
    }).format(number) + " Ã‰ÌÂ";
}

/*  ÃÂÌ“ —ﬁ„ Ê« ”«» */

function getWhatsAppURL(data) {
    const phone = data.whatsapp.replace(/\D/g, "");

    if (!/^\d{8,15}$/.test(phone)) {
        return "";
    }

    const text = data.ctaMessage ||
        `„—Õ»«° √—Ìœ «·«” ›”«— ⁄‰ ${data.headline}.`;

    return "https://wa.me/" + phone +
        "?text=" + encodeURIComponent(text);
}

/*  ÃÂÌ“ ﬁ«∆„… «·„„Ì“«  */

function buildFeatures(features) {
    const items = features
        .split(/\r?\n/)
        .map(item => item.trim())
        .filter(Boolean)
        .slice(0, 30);

    if (items.length === 0) {
        return "";
    }

    return `
        <section class="features-section">
            <h2>·„«–«  Œ «—‰«ø</h2>
            <div class="features-grid">
                ${items.map(item => `
                    <div class="feature">
                        <span class="check">?</span>
                        <span>${escapeHTML(item)}</span>
                    </div>
                `).join("")}
            </div>
        </section>
    `;
}

function saveProject() {

        const project = buildSavedProject();

        showMessage(" „ Õ›Ÿ «·„‘—Ê⁄ ›Ì Â–« «·„ ’›Õ.");
    } catch (error) {

/* «” ⁄«œ… «·„‘—Ê⁄ */

function restoreProject() {
    try {
        const raw = localStorage.getItem(STORAGE_KEY);

        if (!raw) {
            return false;
        }

        const project = JSON.parse(raw);

        if (!project || !project.data) {
            return false;
        }

        const allowedTemplates = [
            "product",
            "service",
            "restaurant"
        ];

        const data = project.data;

        for (const [key, value] of Object.entries(defaults)) {
            const element = $(key);

            if (!element) {
                continue;
            }

            if (key === "template") {
                element.value = allowedTemplates.includes(data[key])
                    ? data[key]
                    : defaults[key];
            } else if (key === "primaryColor" ||
                       key === "backgroundColor") {
                element.value = safeColor(value === data[key] ? value : data[key], defaults[key]);
            } else if (typeof data[key] === "string") {
                element.value = data[key];
            }
        }

        productImage = typeof project.image === "string" &&
            project.image.startsWith("data:image/") &&
            project.image.length < 5_000_000
                ? project.image
                : "";

        $("removeImage").hidden = !productImage;

        updatePreview();

        return true;
    } catch (error) {
        console.error("Restore failed:", error);
        return false;
    }
}

/*  ‰“Ì· ’›Õ… «·Â»Êÿ ﬂ„·› HTML „” ﬁ· */

function downloadLanding() {
    const data = getFormData();
    const error = validateData(data);

    if (error) {
        showMessage(error, true);
        return;
    }

    const html = generateLandingHTML(data, productImage);

    const blob = new Blob(
        ["<!DOCTYPE html>\n", html],
        { type: "text/html;charset=utf-8" }
    );

    const url = URL.createObjectURL(blob);

    const link = document.createElement("a");

    link.href = url;
    link.download = "landing-page.html";

    document.body.appendChild(link);
    link.click();
    link.remove();

    setTimeout(() => URL.revokeObjectURL(url), 1500);

    showMessage(
        " „  ‰“Ì· ’›Õ… «·Â»Êÿ. «› Õ „·› HTML ›Ì «·„ ’›Õ · Ã—» Â«."
    );
}

/*  Õ„Ì· «·’Ê—… «·„Œ «—… */

$("imageUpload").addEventListener("change", (event) => {
    const file = event.target.files?.[0];

    if (!file) {
        return;
    }

    const allowedTypes = [
        "image/jpeg",
        "image/png",
        "image/webp"
    ];

    if (!allowedTypes.includes(file.type)) {
        showMessage("«Œ — ’Ê—… »’Ì€… JPG √Ê PNG √Ê WebP.", true);
        event.target.value = "";
        return;
    }

    if (file.size > 3 * 1024 * 1024) {
        showMessage(
            "ÕÃ„ «·’Ê—… √ﬂ»— „‰ 3 „ÌÃ«»«Ì . «Œ — ’Ê—… √’€—.",
            true
        );
        event.target.value = "";
        return;
    }

    const reader = new FileReader();

    reader.onload = () => {
        if (typeof reader.result !== "string" ||
            !reader.result.startsWith("data:image/")) {
            showMessage(" ⁄–— ﬁ—«¡… «·’Ê—….", true);
            return;
        }

        productImage = reader.result;
        $("removeImage").hidden = false;

        updatePreview();
        showMessage(" „  ≈÷«›… «·’Ê—… »‰Ã«Õ.");
    };

    reader.onerror = () => {
        showMessage("ÕœÀ Œÿ√ √À‰«¡ ﬁ—«¡… «·’Ê—….", true);
    };

    reader.readAsDataURL(file);
});
